package com.evision

import io.flutter.embedding.android.FlutterActivity

import android.content.Intent
import android.os.Bundle
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel

class MainActivity : FlutterActivity() {

    private val CHANNEL = "app.channel.shared.data"

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, CHANNEL).setMethodCallHandler { call, result ->
            if (call.method == "getSharedData") {
                val intentData: String? = intent?.dataString

                // Clear the intent data after use
                intent?.data = null

                result.success(intentData)
            } else {
                result.notImplemented()
            }
        }
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        this.intent = intent
        val data = intent.dataString

        // Pass the deep link to Flutter via the MethodChannel
        val engine = flutterEngine ?: return // Handle null safety for `flutterEngine`
        MethodChannel(engine.dartExecutor.binaryMessenger, CHANNEL)
            .invokeMethod("getSharedData", data)
    }
}

import UIKit
import Flutter

@UIApplicationMain
@objc class AppDelegate: FlutterAppDelegate {
    override func application(
        _ application: UIApplication,
        continue userActivity: NSUserActivity,
        restorationHandler: @escaping ([UIUserActivityRestoring]?) -> Void
    ) -> Bool {
        if let url = userActivity.webpageURL {
            let flutterChannel = FlutterMethodChannel(name: "app.channel.shared.data",
                                                       binaryMessenger: window?.rootViewController as! FlutterBinaryMessenger)
            flutterChannel.invokeMethod("getSharedData", arguments: url.absoluteString)
        }
        return super.application(application, continue: userActivity, restorationHandler: restorationHandler)
    }
}
